function splitCsv(value?: string): string[] {
  return (value ?? '')
    .split(',')
    .map((s) => s.trim())
    .filter(Boolean);
}

/** '{"os":"macOS"}' -> Prisma JSON path filters (equality only). */
function parseJsonFilter(raw: string | undefined, name: string) {
  if (!raw) return [];
  let parsed: unknown;
  try {
    parsed = JSON.parse(raw);
  } catch {
    throw new BadRequestException(`${name} must be a valid JSON object`);
  }
  const result = jsonFilterSchema.safeParse(parsed);
  if (!result.success) {
    throw new BadRequestException(
      `${name} must be a flat JSON object of string, number or boolean values`,
    );
  }
  return Object.entries(result.data).map(([key, value]) => ({ path: [key], equals: value }));
}

@Injectable()
export class ProductsService {
  constructor(
    private readonly prisma: PrismaService,
    private readonly media: MediaService,
  ) {}

  async list(query: ListProductsDto) {
    const { page, pageSize, sort } = query;
    const { where, variantWhere } = this.buildWhere(query);
    const include = productInclude(variantWhere);
    const meta = (total: number) => ({
      page,
      pageSize,
      total,
      totalPages: Math.ceil(total / pageSize),
    });

    if (sort === 'price_asc' || sort === 'price_desc') {
      // Prisma cannot order by MIN(variant price), so rank ids first (lightweight), then hydrate one page.
      // Fine for a few thousand products; denormalize min price later if the catalog grows.
      const rows = await this.prisma.product.findMany({
        where,
        select: { id: true, variants: { where: variantWhere, select: { priceMinor: true } } },
      });
      const dir = sort === 'price_asc' ? 1 : -1;
      const ids = rows
        .map((r) => ({ id: r.id, min: Math.min(...r.variants.map((v) => v.priceMinor)) }))
        .sort((a, b) => dir * (a.min - b.min) || a.id.localeCompare(b.id))
        .slice((page - 1) * pageSize, page * pageSize)
        .map((r) => r.id);
      const products = await this.prisma.product.findMany({ where: { id: { in: ids } }, include });
      const byId = new Map(products.map((p) => [p.id, p]));
      const items = ids.map((id) => byId.get(id)).filter((p): p is ProductRow => p !== undefined);
      return { items: items.map(mapProductStatic), meta: meta(rows.length) };
    }

    const [total, rows] = await this.prisma.$transaction([
      this.prisma.product.count({ where }),
      this.prisma.product.findMany({
        where,
        include,
        orderBy: ORDER_BY[sort],
        skip: (page - 1) * pageSize,
        take: pageSize,
      }),
    ]);
    return { items: rows.map(mapProductStatic), meta: meta(total) };
  }

  async getBySlug(slug: string) {
    const product = await this.prisma.product.findFirst({
      where: { slug, isActive: true },
      include: productInclude({ isActive: true }),
    });
    if (!product) throw new NotFoundException('Product not found');
    return mapProductDetailStatic(product);
  }

  async create(dto: CreateProductDto, files?: Express.Multer.File[]) {
    // Validate everything that can be validated BEFORE uploading anything
    const category = await this.getCategory(dto.categoryId);
    await this.assertBrandExists(dto.brandId);
    const specs = this.validateSpecs(category, dto.specs);

    return this.media.withUploads(files, PRODUCT_IMAGE_FOLDER, async (images) => {
      const product = await this.prisma.product.create({
        data: {
          name: dto.name,
          slug: resolveSlug(dto.name, dto.slug),
          description: dto.description,
          brandId: dto.brandId,
          categoryId: dto.categoryId,
          specs,
          warrantyMonths: dto.warrantyMonths,
          isActive: dto.isActive,

              secureUrl: img.secureUrl,
              publicId: img.publicId,

        include: productInclude({}),
      });
      return mapProductDetail(product);
    });
  }

  async update(id: string, dto: UpdateProductDto, files?: Express.Multer.File[]) {
    const { removeImageIds = [], primaryImageId, ...fields } = dto;

    const existing = await this.prisma.product.findUnique({
      where: { id },
      select: {
        categoryId: true,
        specs: true,
        // Variant images are managed through the variant endpoints
        images: {
          where: { variantId: null },
          orderBy: { position: 'asc' },
          select: { id: true, publicId: true, isPrimary: true, position: true },
        },
      },
    });
    if (!existing) throw new NotFoundException('Product not found');

    const hasFieldChange = Object.values(fields).some((v) => v !== undefined);
    if (!hasFieldChange && !removeImageIds.length && !primaryImageId && !files?.length) {
      throw new BadRequestException('Nothing to update');
    }
    if (fields.brandId) await this.assertBrandExists(fields.brandId);

    // Re-validate specs whenever the specs or the category change (specs are replaced, not merged)
    let specs: Prisma.InputJsonObject | undefined;
    if (fields.specs !== undefined || fields.categoryId !== undefined) {
      const category = await this.getCategory(fields.categoryId ?? existing.categoryId);
      specs = this.validateSpecs(category, fields.specs ?? existing.specs);
    }

    const knownIds = new Set(existing.images.map((i) => i.id));
    for (const imageId of [...removeImageIds, ...(primaryImageId ? [primaryImageId] : [])]) {
      if (!knownIds.has(imageId)) {
        throw new BadRequestException(`Image ${imageId} does not belong to this product`);
      }
    }
    if (primaryImageId && removeImageIds.includes(primaryImageId)) {
      throw new BadRequestException('primaryImageId cannot also be removed');
    }

    const removed = existing.images.filter((i) => removeImageIds.includes(i.id));
    const remaining = existing.images.filter((i) => !removeImageIds.includes(i.id));
    if (remaining.length + (files?.length ?? 0) > MAX_IMAGES_PER_PRODUCT) {
      throw new BadRequestException(`A product can have at most ${MAX_IMAGES_PER_PRODUCT} images`);
    }

    const product = await this.media.withUploads(files, PRODUCT_IMAGE_FOLDER, (uploaded) => {
      // Keep the requested/current primary; if none is left, the first new image becomes primary
      const primaryId =
        primaryImageId ?? remaining.find((i) => i.isPrimary)?.id ?? remaining[0]?.id;
      const nextPosition = remaining.length ? Math.max(...remaining.map((i) => i.position)) + 1 : 0;
      const imagesChanged =
        removed.length > 0 || uploaded.length > 0 || primaryImageId !== undefined;

      return this.prisma.$transaction(async (tx) => {
        if (imagesChanged) {
          if (removed.length) {
            await tx.productImage.deleteMany({ where: { id: { in: removed.map((i) => i.id) } } });
          }
          // The partial unique index allows one primary: unset first, then set
          await tx.productImage.updateMany({
            where: { productId: id, isPrimary: true },
            data: { isPrimary: false },
          });
          if (primaryId) {
            await tx.productImage.update({ where: { id: primaryId }, data: { isPrimary: true } });
          }
          if (uploaded.length) {
            await tx.productImage.createMany({
              data: uploaded.map((img, i) => ({
                productId: id,
                secureUrl: img.secureUrl,
                publicId: img.publicId,
                position: nextPosition + i,
                isPrimary: primaryImageId === undefined && i === 0,
              })),
            });
          }
        }
        // Last, so the returned product includes the final images
        return tx.product.update({
          where: { id },
          data: { ...fields, specs },
          include: productInclude({}),
        });
      });
    });

    // Only after the DB commit: a failure above must never lose an existing image
    await this.media.deleteAssets(removed.map((i) => i.publicId));
    return mapProductDetail(product);
  }

  async removeImage(productId: string, imageId: string) {
    const image = await this.prisma.productImage.findFirst({
      where: { id: imageId, productId, variantId: null },
      select: { id: true },
    });
    if (!image) throw new NotFoundException('Image not found');
    await this.update(productId, { removeImageIds: [imageId] });
  }

  /** Soft delete: products with sales or stock history are never removed. */
  async deactivate(id: string) {
    await this.prisma.product.update({ where: { id }, data: { isActive: false } });
  }

  private buildWhere(q: ListProductsDto) {
    const variantWhere: Prisma.ProductVariantWhereInput = {
      isActive: true,
      priceMinor: { gte: q.minPrice, lte: q.maxPrice },
      AND: parseJsonFilter(q.attributes, 'attributes').map((attributes) => ({ attributes })),
    };

    const and: Prisma.ProductWhereInput[] = parseJsonFilter(q.specs, 'specs').map((specs) => ({
      specs,
    }));
    const brands = splitCsv(q.brand);
    if (brands.length) and.push({ brand: { slug: { in: brands } } });
    if (q.category) {
      and.push({ category: { OR: [{ slug: q.category }, { parent: { slug: q.category } }] } });
    }
    if (q.q) {
      and.push({
        OR: [
          { name: { contains: q.q, mode: 'insensitive' } },
          { brand: { name: { contains: q.q, mode: 'insensitive' } } },
          { variants: { some: { sku: { contains: q.q, mode: 'insensitive' } } } },
        ],
      });
    }

    // Public listing: active product with at least one active variant matching the variant filters
    const where: Prisma.ProductWhereInput = {
      isActive: true,
      variants: { some: variantWhere },
      AND: and,
    };
    return { where, variantWhere };
  }

  private async getCategory(id: string) {
    const category = await this.prisma.category.findUnique({
      where: { id },
      select: { slug: true, parent: { select: { slug: true } } },
    });
    if (!category) throw new BadRequestException('categoryId does not exist');
    return category;
  }

  private async assertBrandExists(id: string) {
    if ((await this.prisma.brand.count({ where: { id } })) === 0) {
      throw new BadRequestException('brandId does not exist');
    }
  }

  private validateSpecs(
    category: { slug: string; parent: { slug: string } | null },
    specs: unknown,
  ): Prisma.InputJsonObject {
    const result = getSpecsSchema(category.slug, category.parent?.slug).safeParse(specs);
    if (!result.success) {
      throw new BadRequestException({
        statusCode: 400,
        error: 'Bad Request',
        message: `Invalid specs for category "${category.slug}"`,
        errors: result.error.issues.map((i) => ({ path: i.path.join('.'), message: i.message })),
      });
    }
    return result.data as Prisma.InputJsonObject;
  }
}
