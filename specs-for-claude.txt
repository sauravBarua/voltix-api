import { z } from 'zod';

const laptopSpecs = z.strictObject({
  cpu: z.string().min(1),
  gpu: z.string().min(1).optional(),
  screenSizeInch: z.number().positive().max(30),
  resolution: z.string().min(1).optional(),
  batteryWh: z.number().positive().optional(),
  weightKg: z.number().positive().optional(),
  os: z.string().min(1),
});

const smartphoneSpecs = z.strictObject({
  chipset: z.string().min(1),
  screenSizeInch: z.number().positive().max(15),
  batteryMah: z.number().int().positive(),
  rearCameraMp: z.number().positive().optional(),
  os: z.string().min(1),
  fiveG: z.boolean().optional(),
});

// Accessories are heterogeneous: only a "type" is required
const accessorySpecs = z.looseObject({ type: z.string().min(1) });

const genericSpecs = z.record(z.string(), z.unknown());

const byCategorySlug: Record<string, z.ZodType> = {
  laptops: laptopSpecs,
  smartphones: smartphoneSpecs,
  accessories: accessorySpecs,
};

/** Tries the category slug, then its parent's slug, then falls back to a generic object. */
export function getSpecsSchema(...slugs: (string | null | undefined)[]): z.ZodType {
  for (const slug of slugs) {
    if (slug && Object.hasOwn(byCategorySlug, slug)) return byCategorySlug[slug];
  }
  return genericSpecs;
}
import { SetMetadata, applyDecorators } from '@nestjs/common';
import { ApiBearerAuth, ApiForbiddenResponse, ApiUnauthorizedResponse } from '@nestjs/swagger';
import type { Role } from '../generated/prisma/client';

export const ACCESS_KEY = 'access';
export const SKIP_AUDIT_KEY = 'skipAudit';
export type Access = 'public' | readonly Role[];
export const ALL_ROLES: Role[] = ['CUSTOMER', 'INVENTORY_OPERATOR', 'ADMIN'];

/** No authentication. Use sparingly: health, public catalog reads, auth entry points, the Stripe webhook. */
export const Public = () => SetMetadata(ACCESS_KEY, 'public');

/** Logged-in user with one of these roles. A handler-level policy overrides the class-level one. */
export const Roles = (...roles: Role[]) =>
  applyDecorators(
    SetMetadata(ACCESS_KEY, roles),
    ApiBearerAuth(),
    ApiUnauthorizedResponse({ description: 'Missing, invalid or expired access token' }),
    ApiForbiddenResponse({ description: 'Your role cannot access this resource' }),
  );

/** Any logged-in user. */
export const Authenticated = () => Roles(...ALL_ROLES);

/** The handler writes its own, richer audit entry. */
export const SkipAudit = () => SetMetadata(SKIP_AUDIT_KEY, true);
import { createZodDto } from 'nestjs-zod';
import { z } from 'zod';

export const slugSchema = z
  .string()
  .min(2)
  .max(100)
  .regex(/^[a-z0-9]+(?:-[a-z0-9]+)*$/, 'lowercase letters, digits and hyphens only');

/** PATCH bodies must change something. */
export const hasAtLeastOneField = (o: Record<string, unknown>) =>
  Object.values(o).some((v) => v !== undefined);

export class IdParamDto extends createZodDto(z.object({ id: z.uuid() })) {}
export class SlugParamDto extends createZodDto(z.object({ slug: slugSchema })) {}
